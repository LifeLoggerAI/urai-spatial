```tsx
'use client'

import { useEffect, useState } from 'react'
import { gsap } from 'gsap'

export default function TransitionOverlay({ trigger }: { trigger: boolean }) {
  const [visible, setVisible] = useState(false)

  useEffect(() => {
    if (trigger) {
      setVisible(true)
      gsap.fromTo(
        '#overlay',
        { opacity: 0 },
        {
          opacity: 1,
          duration: 0.3,
          yoyo: true,
          repeat: 1,
          onComplete: () => setVisible(false),
        }
      )
    }
  }, [trigger])

  if (!visible) return null

  return (
    <div
      id="overlay"
      style={{
        position: 'fixed',
        inset: 0,
        background: 'white',
        pointerEvents: 'none',
        zIndex: 1000,
        opacity: 0,
      }}
    />
  )
}
```

```ts
'use client'

import { useRef } from 'react'

export function useTransitionGuard() {
  const isTransitioning = useRef(false)

  const beginTransition = (): boolean => {
    if (isTransitioning.current) return false
    isTransitioning.current = true
    return true
  }

  const endTransition = () => {
    isTransitioning.current = false
  }

  return {
    beginTransition,
    endTransition,
    isTransitioning,
  }
}
```

```ts
'use client'

import { gsap } from 'gsap'
import { Camera } from 'three'

export function animateCamera(
  camera: Camera,
  targetPosition: { x: number; y: number; z: number },
  duration: number
): Promise<void> {
  return new Promise((resolve) => {
    const timeline = gsap.timeline({
      onComplete: () => {
        timeline.kill()
        resolve()
      },
    })

    timeline.to(camera.position, {
      x: targetPosition.x,
      y: targetPosition.y,
      z: targetPosition.z,
      duration,
      ease: 'power2.inOut',
    })
  })
}
```

This guarantees:

* No early route push
* No dangling timeline
* Clean sequencing

```tsx
'use client'

import { useMemo, useRef } from 'react'
import { useFrame } from '@react-three/fiber'
import { Mesh, Color } from 'three'
import { useRouter } from 'next/navigation'
import { useTransitionGuard } from '@/lib/useTransitionGuard'
import { animateCamera } from '@/lib/cameraAnimator'

interface MemoryStar {
  id: number
  position: [number, number, number]
  intensity: number
  valence: number
}

export default function Starfield() {
  const router = useRouter()
  const { beginTransition, endTransition } = useTransitionGuard()

  const stars = useMemo<MemoryStar[]>(() => {
    return Array.from({ length: 30 }).map((_, index) => ({
      id: index,
      position: [
        (Math.random() - 0.5) * 20,
        (Math.random() - 0.5) * 10,
        (Math.random() - 0.5) * 20,
      ],
      intensity: Math.random(),
      valence: Math.random() * 2 - 1,
    }))
  }, [])

  const starRefs = useRef<Mesh[]>([])

  useFrame(({ clock }) => {
    const t = clock.elapsedTime

    starRefs.current.forEach((mesh, index) => {
      if (!mesh) return

      const star = stars[index]

      // Pulse speed scaled by intensity
      const pulseSpeed = 1 + star.intensity * 2
      const pulse = 1 + Math.sin(t * pulseSpeed) * 0.05 * star.intensity

      // Scale based on intensity
      const baseScale = 1 + star.intensity * 0.4
      mesh.scale.setScalar(baseScale * pulse)

      // Emissive intensity mapping
      const material = mesh.material
      if ('emissiveIntensity' in material) {
        const emissiveBase = 0.5 + star.intensity * 1.5
        material.emissiveIntensity = emissiveBase * pulse
      }
    })
  })

  const handleStarClick = async (star: MemoryStar) => {
    if (!beginTransition()) return

    const cameraTarget = {
      x: star.position[0],
      y: star.position[1],
      z: star.position[2] + 2,
    }

    await animateCamera((window as any).__r3fCamera, cameraTarget, 1)

    endTransition()
    router.push('/replay/1')
  }

  return (
    <>
      {stars.map((star, index) => {
        // Map valence to subtle color shift
        const baseColor = new Color(0xffffff)

        const cool = new Color(0x88bbff)
        const warm = new Color(0xffd4aa)

        const finalColor =
          star.valence >= 0
            ? baseColor.clone().lerp(warm, star.valence * 0.3)
            : baseColor.clone().lerp(cool, Math.abs(star.valence) * 0.3)

        return (
          <mesh
            key={star.id}
            ref={(el) => {
              if (el) starRefs.current[index] = el
            }}
            position={star.position}
            onClick={() => handleStarClick(star)}
          >
            <sphereGeometry args={[0.15, 16, 16]} />
            <meshStandardMaterial
              color={finalColor}
              emissive={finalColor}
              emissiveIntensity={0.5}
            />
          </mesh>
        )
      })}
    </>
  )
}
```

```ts
// lib/emotionalToneEngine.ts

export interface MemoryStar {
  intensity: number
  valence: number
}

export interface EmotionalTone {
  ambientIntensityMultiplier: number
  hueShift: number
  particleDriftMultiplier: number
  audioProfile: 'calm' | 'tense' | 'warm'
}

function clamp(value: number, min: number, max: number): number {
  return Math.min(Math.max(value, min), max)
}

export function computeEmotionalTone(
  stars: MemoryStar[]
): EmotionalTone {
  if (stars.length === 0) {
    return {
      ambientIntensityMultiplier: 1,
      hueShift: 0,
      particleDriftMultiplier: 1,
      audioProfile: 'calm',
    }
  }

  const totalIntensity = stars.reduce(
    (sum, star) => sum + clamp(star.intensity, 0, 1),
    0
  )

  const totalValence = stars.reduce(
    (sum, star) => sum + clamp(star.valence, -1, 1),
    0
  )

  const avgIntensity = clamp(
    totalIntensity / stars.length,
    0,
    1
  )

  const avgValence = clamp(
    totalValence / stars.length,
    -1,
    1
  )

  // Subtle scaling
  const ambientIntensityMultiplier =
    1 + avgIntensity * 0.2

  const hueShift =
    avgValence * 0.1

  const particleDriftMultiplier =
    0.8 + avgIntensity * 0.5

  let audioProfile: 'calm' | 'tense' | 'warm' = 'calm'

  if (avgValence > 0.25) {
    audioProfile = 'warm'
  } else if (avgValence < -0.25) {
    audioProfile = 'tense'
  }

  return {
    ambientIntensityMultiplier,
    hueShift,
    particleDriftMultiplier,
    audioProfile,
  }
}
```

```tsx
import { ReactNode, useEffect, useRef } from 'react'
import { Canvas, useFrame, useThree } from '@react-three/fiber'
import { PerspectiveCamera } from '@react-three/drei'
import { AmbientLight, Color } from 'three'
import { computeEmotionalTone, MemoryStar } from '@/lib/emotionalToneEngine'

interface SceneManagerProps {
  children: ReactNode
  stars?: MemoryStar[]
  sceneType?: 'home' | 'lifemap' | 'replay'
}

function LightingController({
  stars = [],
  sceneType = 'home',
}: {
  stars?: MemoryStar[]
  sceneType?: 'home' | 'lifemap' | 'replay'
}) {
  const ambientRef = useRef<AmbientLight>(null)
  const toneRef = useRef(computeEmotionalTone(stars))

  useEffect(() => {
    toneRef.current = computeEmotionalTone(stars)
  }, [stars])

  useFrame(({ clock }) => {
    if (!ambientRef.current) return

    const t = clock.elapsedTime
    const tone = toneRef.current

    // Base intensity
    let baseIntensity = 0.6

    if (sceneType === 'lifemap') baseIntensity = 0.65
    if (sceneType === 'replay') baseIntensity = 0.55

    // Subtle breathing
    const breath =
      Math.sin(t * 0.5) *
      0.05 *
      tone.ambientIntensityMultiplier

    ambientRef.current.intensity = baseIntensity + breath

    // Hue shift logic (very subtle)
    const baseColor =
      sceneType === 'replay'
        ? new Color('#4a00ff')
        : new Color('#ffffff')

    const shifted = baseColor.clone().offsetHSL(
      tone.hueShift,
      0,
      0
    )

    ambientRef.current.color = shifted
  })

  return <ambientLight ref={ambientRef} intensity={0.6} />
}

export default function SceneManager({
  children,
  stars,
  sceneType = 'home',
}: SceneManagerProps) {
  return (
    <Canvas shadows>
      <PerspectiveCamera makeDefault position={[0, 2, 8]} />
      <LightingController stars={stars} sceneType={sceneType} />
      <directionalLight position={[5, 10, 5]} intensity={1} />
      {children}
    </Canvas>
  )
}
```

```tsx
'use client'

import { useMemo, useRef } from 'react'
import { useFrame } from '@react-three/fiber'
import { Points, BufferGeometry, Float32BufferAttribute } from 'three'

interface AtmosphereParticlesProps {
  count?: number
  spread?: number
  driftMultiplier?: number
}

export default function AtmosphereParticles({
  count = 120,
  spread = 30,
  driftMultiplier = 1,
}: AtmosphereParticlesProps) {
  const pointsRef = useRef<Points>(null)

  const positions = useMemo(() => {
    const arr = new Float32Array(count * 3)

    for (let i = 0; i < count; i++) {
      arr[i * 3] = (Math.random() - 0.5) * spread
      arr[i * 3 + 1] = (Math.random() - 0.5) * spread
      arr[i * 3 + 2] = (Math.random() - 0.5) * spread
    }

    return arr
  }, [count, spread])

  const geometry = useMemo(() => {
    const geo = new BufferGeometry()
    geo.setAttribute('position', new Float32BufferAttribute(positions, 3))
    return geo
  }, [positions])

  useFrame(({ clock }) => {
    if (!pointsRef.current) return

    const t = clock.elapsedTime
    const pos = geometry.attributes.position as Float32BufferAttribute

    for (let i = 0; i < count; i++) {
      const yIndex = i * 3 + 1
      pos.array[yIndex] += 0.0005 * driftMultiplier

      // Soft reset if too high
      if (pos.array[yIndex] > spread / 2) {
        pos.array[yIndex] = -spread / 2
      }
    }

    pos.needsUpdate = true

    // subtle opacity pulse
    const material = pointsRef.current.material
    if ('opacity' in material) {
      material.opacity =
        0.08 + Math.sin(t * 0.3) * 0.02
    }
  })

  return (
    <points ref={pointsRef} geometry={geometry}>
      <pointsMaterial
        size={0.1}
        transparent
```

# STEP 1 — Replace audioManager.ts

Open:

```
lib/audioManager.ts
```

Replace with this:

```ts
'use client'

import { gsap } from 'gsap'

class AudioManager {
  private currentAudio: HTMLAudioElement | null = null
  private hasInteracted = false

  registerInteraction() {
    this.hasInteracted = true
  }

  async play(src: string, fadeDuration = 1) {
    if (!this.hasInteracted) return

    if (this.currentAudio) {
      await this.fadeOut(fadeDuration)
    }

    const audio = new Audio(src)
    audio.loop = true
    audio.volume = 0

    try {
      await audio.play()
    } catch {
      return
    }

    this.currentAudio = audio

    gsap.to(audio, {
      volume: 0.2,
      duration: fadeDuration,
      ease: 'power2.out',
    })
  }

  async fadeOut(duration = 1) {
    if (!this.currentAudio) return

    const audio = this.currentAudio

    return new Promise<void>((resolve) => {
      gsap.to(audio, {
        volume: 0,
        duration,
        ease: 'power2.inOut',
        onComplete: () => {
          audio.pause()
          this.currentAudio = null
          resolve()
        },
      })
    })
  }

  stopImmediate() {
    if (!this.currentAudio) return
    this.currentAudio.pause()
    this.currentAudio = null
  }
}

export const audioManager = new AudioManager()
```

## 1️⃣ ErrorBoundary Wrapper

Create:

`components/ErrorBoundary.tsx`

```tsx
'use client'

import React from 'react'

interface Props {
  children: React.ReactNode
}

interface State {
  hasError: boolean
}

export default class ErrorBoundary extends React.Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false }
  }

  static getDerivedStateFromError(): State {
    return { hasError: true }
  }

  componentDidCatch(error: Error) {
    console.error('Spatial Error:', error)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{
          width: '100vw',
          height: '100vh',
          background: '#000',
          color: '#fff',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center'
        }}>
          Spatial encountered an error.
        </div>
      )
    }

    return this.props.children
  }
}
```

Wrap `<Canvas>` or root layout:

```tsx
<ErrorBoundary>
  <Canvas>...</Canvas>
</ErrorBoundary>
```

Now scene crash ≠ app crash.

## 2️⃣ Visibility Pause / Resume Handler

Create:

`lib/visibilityManager.ts`

```ts
'use client'

let isPaused = false

export function setupVisibilityHandler(
  onPause: () => void,
  onResume: () => void
) {
  document.addEventListener('visibilitychange', () => {
    if (document.hidden && !isPaused) {
      isPaused = true
      onPause()
    } else if (!document.hidden && isPaused) {
      isPaused = false
      onResume()
    }
  })
}
```
