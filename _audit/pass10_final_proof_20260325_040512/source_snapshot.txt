function clamp01(v: number) {
  return Math.max(0, Math.min(1, v));
}

function easeIn(t: number) {
  const x = clamp01(t);
  return x * x * x;
}

function easeOut(t: number) {
  const x = clamp01(t);
  return 1 - Math.pow(1 - x, 3);
}

function easeInOut(t: number) {
  const x = clamp01(t);
  return x * x * (3 - 2 * x);
}

function heldRelease(t: number) {
  if (t <= 0.08) return t * 0.18;
  return 0.0144 + easeInOut((t - 0.08) / 0.92) * 0.9856;
}

function gravitationalPull(t: number) {
  const x = clamp01(t);
  return easeInOut(Math.pow(x, 0.82));
}

function resurfacing(t: number) {
  const x = clamp01(t);
  return 0.35 * easeIn(x) + 0.65 * easeOut(x);
}

function releaseEase(t: number) {
  const x = clamp01(t);
  if (x < 0.18) return easeOut(x / 0.18) * 0.28;
  return 0.28 + easeInOut((x - 0.18) / 0.82) * 0.72;
}
