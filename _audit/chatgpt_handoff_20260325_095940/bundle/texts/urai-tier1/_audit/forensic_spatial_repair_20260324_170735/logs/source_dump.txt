    32	function seededRandom(seed: number) {
    33	  let t = seed >>> 0;
    34	  return () => {
    35	    t += 0x6d2b79f5;
    36	    let r = Math.imul(t ^ (t >>> 15), 1 | t);
    37	    r ^= r + Math.imul(r ^ (r >>> 7), 61 | r);
    38	    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
    39	  };
    40	}

    42	function clamp01(v: number) {
    43	  if (v < 0) return 0;
    44	  if (v > 1) return 1;
    45	  return v;
    46	}

    48	function easeInOutCubic(t: number) {
    49	  const x = clamp01(t);
    50	  return x < 0.5 ? 4 * x * x * x : 1 - Math.pow(-2 * x + 2, 3) / 2;
    51	}
    52	
    53	function easeOutCubic(t: number) {
    54	  const x = clamp01(t);
    55	  return 1 - Math.pow(1 - x, 3);
    56	}

    58	function mix(a: number, b: number, t: number) {
    59	  return a + (b - a) * t;

    66	    mix(a.z, b.z, t)

   126	function getTransitionDuration(mode: TransitionMode) {
   127	  if (mode === "homeToLifemap" || mode === "lifemapToHome") return 3.0;
   128	  if (mode === "lifemapToFocus" || mode === "focusToLifemap") return 1.55;
   129	  if (mode === "focusToReplay" || mode === "replayToFocus") return 1.15;

   133	function computeTransitionPose(
   134	  mode: TransitionMode,
   135	  progress: number,
   136	  selected: StarPoint | null
   137	) {
   138	  const home = getHomePose();
   139	  const lifemap = getLifemapPose();
   140	
   141	  if (mode === "homeToLifemap") {
   142	    const p = clamp01(progress);
   143	
   144	    const tiltT = easeOutCubic(clamp01(p / 0.22));
   145	    const liftT = easeInOutCubic(clamp01((p - 0.12) / 0.28));
   146	    const travelT = easeInOutCubic(clamp01((p - 0.34) / 0.66));
   147	
   148	    const tiltPos = new THREE.Vector3(0, 1.90, 7.2);
   149	    const tiltLook = new THREE.Vector3(0, 3.1, -16.5);
   150	
   151	    const liftPos = new THREE.Vector3(0, 1.35, 3.6);
   152	    const liftLook = new THREE.Vector3(0, 1.4, -27.5);
   153	
   154	    const phase1Pos = mixVec3(home.pos, tiltPos, tiltT);
   155	    const phase1Look = mixVec3(home.look, tiltLook, tiltT);
   156	
   157	    const phase2Pos = mixVec3(phase1Pos, liftPos, liftT);
   158	    const phase2Look = mixVec3(phase1Look, liftLook, liftT);
   159	
   160	    return {
   161	      pos: mixVec3(phase2Pos, lifemap.pos, travelT),
   162	      look: mixVec3(phase2Look, lifemap.look, travelT),
   163	      fov: mix(42, 49, travelT),
   164	    };
   165	  }
   166	
   167	  if (mode === "lifemapToHome") {
   168	    const p = clamp01(progress);
   169	
   170	    const travelT = easeInOutCubic(clamp01(p / 0.46));
   171	    const settleT = easeInOutCubic(clamp01((p - 0.38) / 0.62));
   172	
   173	    const liftPos = new THREE.Vector3(0, 1.35, 3.6);
   174	    const liftLook = new THREE.Vector3(0, 1.4, -27.5);
   175	
   176	    const tiltPos = new THREE.Vector3(0, 1.90, 7.2);
   177	    const tiltLook = new THREE.Vector3(0, 3.1, -16.5);
   178	
   179	    const phase1Pos = mixVec3(lifemap.pos, liftPos, travelT);
   180	    const phase1Look = mixVec3(lifemap.look, liftLook, travelT);
   181	
   182	    const phase2Pos = mixVec3(phase1Pos, tiltPos, settleT);
   183	    const phase2Look = mixVec3(phase1Look, tiltLook, settleT);
   184	
   185	    return {
   186	      pos: mixVec3(phase2Pos, home.pos, settleT),
   187	      look: mixVec3(phase2Look, home.look, settleT),
   188	      fov: mix(49, 42, settleT),
   189	    };
   190	  }
   191	
   192	  if (selected && mode === "lifemapToFocus") {
   193	    const focus = getFocusPose(selected);
   194	    const t = easeInOutCubic(progress);
   195	    return {
   196	      pos: mixVec3(getLifemapPose().pos, focus.pos, t),
   197	      look: mixVec3(getLifemapPose().look, focus.look, t),
   198	      fov: mix(49, 38, t),
   199	    };
   200	  }
   201	
   202	  if (selected && mode === "focusToLifemap") {
   203	    const focus = getFocusPose(selected);
   204	    const t = easeInOutCubic(progress);
   205	    return {
   206	      pos: mixVec3(focus.pos, getLifemapPose().pos, t),
   207	      look: mixVec3(focus.look, getLifemapPose().look, t),
   208	      fov: mix(38, 49, t),
   209	    };
   210	  }
   211	
   212	  if (selected && mode === "focusToReplay") {
   213	    const focus = getFocusPose(selected);
   214	    const replay = getReplayPose(selected);
   215	
   216	    const p = clamp01(progress);
   217	    const holdT = easeOutCubic(clamp01(p / 0.18));
   218	    const diveT = easeInOutCubic(clamp01((p - 0.12) / 0.88));
   219	
   220	    const prePos = new THREE.Vector3(focus.pos.x, focus.pos.y, focus.pos.z + 0.28);
   221	    const preLook = new THREE.Vector3(focus.look.x, focus.look.y, focus.look.z);
   222	
   223	    const heldPos = mixVec3(focus.pos, prePos, holdT);
   224	    const heldLook = mixVec3(focus.look, preLook, holdT);
   225	
   226	    return {
   227	      pos: mixVec3(heldPos, replay.pos, diveT),
   228	      look: mixVec3(heldLook, replay.look, diveT),
   229	      fov: mix(38, 51, diveT),
   230	    };
   231	  }
   232	
   233	  if (selected && mode === "replayToFocus") {
   234	    const focus = getFocusPose(selected);
   235	    const replay = getReplayPose(selected);
   236	    const t = easeInOutCubic(progress);
   237	    return {
   238	      pos: mixVec3(replay.pos, focus.pos, t),
   239	      look: mixVec3(replay.look, focus.look, t),
   240	      fov: mix(51, 38, t),
   241	    };
   242	  }
   243	
   244	  return null;
   245	}

   247	function CameraRig({
   248	  phase,
   249	  transitionMode,
   250	  transitionProgress,
   251	  selected,
   252	}: {
   253	  phase: ScenePhase;
   254	  transitionMode: TransitionMode;
   255	  transitionProgress: number;
   256	  selected: StarPoint | null;
   257	}) {
   258	  const { camera } = useThree();
   259	
   260	  useFrame((state, delta) => {
   261	    const t = state.clock.elapsedTime;
   262	    const idleX = Math.sin(t * 0.18) * 0.05;
   263	    const idleY = Math.cos(t * 0.22) * 0.03;
   264	
   265	    let targetPos = getHomePose().pos.clone();
   266	    let targetLook = getHomePose().look.clone();
   267	    let targetFov = getHomePose().fov;
   268	
   269	    if (transitionMode !== "idle") {
   270	      const transitionPose = computeTransitionPose(
   271	        transitionMode,
   272	        transitionProgress,
   273	        selected
   274	      );
   275	
   276	      if (transitionPose) {
   277	        targetPos.copy(transitionPose.pos);
   278	        targetLook.copy(transitionPose.look);
   279	        targetFov = transitionPose.fov;
   280	      }
   281	    } else if (phase === "home") {
   282	      const pose = getHomePose();
   283	      targetPos.copy(pose.pos);
   284	      targetLook.copy(pose.look);
   285	      targetFov = pose.fov;
   286	    } else if (phase === "lifemap") {
   287	      const pose = getLifemapPose();
   288	      targetPos.copy(pose.pos);
   289	      targetLook.copy(pose.look);
   290	      targetFov = pose.fov;
   291	    } else if (phase === "focus" && selected) {
   292	      const pose = getFocusPose(selected);
   293	      targetPos.copy(pose.pos);
   294	      targetLook.copy(pose.look);
   295	      targetFov = pose.fov;
   296	    } else if (phase === "replay" && selected) {
   297	      const pose = getReplayPose(selected);
   298	      targetPos.copy(pose.pos);
   299	      targetLook.copy(pose.look);
   300	      targetFov = pose.fov;
   301	    }
   302	
   303	    const idleScale = phase === "home" && transitionMode === "idle" ? 1 : 0.16;
   304	    targetPos.x += idleX * idleScale;
   305	    targetPos.y += idleY * 0.65 * idleScale;
   306	    targetLook.x += idleX * 0.3 * idleScale;
   307	    targetLook.y += idleY * 0.25 * idleScale;
   308	
   309	    const lerp = 1 - Math.exp(-delta * 4.2);
   310	    camera.position.lerp(targetPos, lerp);
   311	    camera.lookAt(targetLook);
   312	
   313	    const perspectiveCamera = camera as PerspectiveCamera;
   314	    perspectiveCamera.fov = mix(
   315	      perspectiveCamera.fov,
   316	      targetFov,
   317	      1 - Math.exp(-delta * 5.0)
   318	    );
   319	    perspectiveCamera.updateProjectionMatrix();
   320	  });
   321	
   322	  return null;
   323	}

   325	function SkyDome({
   326	  phase,
   327	  transitionMode,
   328	  transitionProgress,
   329	}: {
   330	  phase: ScenePhase;
   331	  transitionMode: TransitionMode;
   332	  transitionProgress: number;
   333	}) {
   334	  const materialRef = useRef<THREE.ShaderMaterial | null>(null);
   335	
   336	  useFrame((state) => {
   337	    if (!materialRef.current) return;
   338	    materialRef.current.uniforms.uTime.value = state.clock.elapsedTime;
   339	
   340	    let lift = phase === "home" ? 0 : 1;
   341	    if (transitionMode === "homeToLifemap") {
   342	      lift = easeInOutCubic(transitionProgress);
   343	    } else if (transitionMode === "lifemapToHome") {
   344	      lift = 1 - easeInOutCubic(transitionProgress);
   345	    }
   346	
   347	    materialRef.current.uniforms.uLift.value = lift;
   348	  });
   349	
   350	  return (
   351	    <mesh scale={168}>
   352	      <sphereGeometry args={[1, 48, 48]} />
   353	      <shaderMaterial
   354	        ref={materialRef}
   355	        side={THREE.BackSide}
   356	        depthWrite={false}
   357	        uniforms={{
   358	          uTime: { value: 0 },
   359	          uLift: { value: phase === "home" ? 0 : 1 },
   360	        }}
   361	        vertexShader={`
   362	          varying vec3 vWorld;
   363	          void main() {
   364	            vec4 worldPosition = modelMatrix * vec4(position, 1.0);
   365	            vWorld = worldPosition.xyz;
   366	            gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   367	          }
   368	        `}
   369	        fragmentShader={`
   370	          uniform float uTime;
   371	          uniform float uLift;
   372	          varying vec3 vWorld;
   373	
   374	          void main() {
   375	            vec3 dir = normalize(vWorld);
   376	            float h = clamp(dir.y * 0.5 + 0.5, 0.0, 1.0);
   377	
   378	            vec3 topColor = vec3(0.004, 0.010, 0.030);
   379	            vec3 midColor = vec3(0.008, 0.018, 0.050);
   380	            vec3 lowColor = vec3(0.018, 0.040, 0.082);
   381	
   382	            vec3 base = mix(lowColor, midColor, smoothstep(0.0, 0.55, h));
   383	            base = mix(base, topColor, smoothstep(0.45, 1.0, h));
   384	
   385	            float homeHorizonLift = smoothstep(0.0, 0.22, h) * 0.09 * (1.0 - uLift);
   386	            float lifemapFlatten = smoothstep(0.0, 0.20, h) * 0.052 * uLift;
   387	            float haze = mix(
   388	              smoothstep(0.16, 0.62, h) * 0.11,
   389	              smoothstep(0.0, 1.0, h) * 0.018,
   390	              uLift
   391	            );
   392	            float drift = sin((dir.x + dir.z) * 12.0 + uTime * 0.03) * mix(0.006, 0.0025, uLift);
   393	            float ascentDarken = uLift * smoothstep(0.20, 1.0, h) * 0.10;
   394	
   395	            vec3 color = base + haze + drift + vec3(homeHorizonLift * 0.35, homeHorizonLift * 0.60, homeHorizonLift) - vec3(lifemapFlatten) - ascentDarken;
   396	            gl_FragColor = vec4(color, 1.0);
   397	          }
   398	        `}
   399	      />
   400	    </mesh>
   401	  );
   402	}

   404	function Ground({
   405	  phase,
   406	  transitionMode,
   407	  transitionProgress,
   408	}: {
   409	  phase: ScenePhase;
   410	  transitionMode: TransitionMode;
   411	  transitionProgress: number;
   412	}) {
   413	  const materialRef = useRef<THREE.ShaderMaterial | null>(null);
   414	
   415	  useFrame((state) => {
   416	    if (!materialRef.current) return;
   417	    materialRef.current.uniforms.uTime.value = state.clock.elapsedTime;
   418	
   419	    let visible = phase === "home" ? 1 : 0;
   420	    if (transitionMode === "homeToLifemap") {
   421	      visible = 1 - easeInOutCubic(clamp01(transitionProgress / 0.40));
   422	    } else if (transitionMode === "lifemapToHome") {
   423	      visible = easeInOutCubic(clamp01((transitionProgress - 0.48) / 0.52));
   424	    }
   425	
   426	    materialRef.current.uniforms.uVisible.value = visible;
   427	  });
   428	
   429	  const shouldRender =
   430	    phase === "home" ||
   431	    transitionMode === "homeToLifemap" ||
   432	    transitionMode === "lifemapToHome";
   433	
   434	  if (!shouldRender) return null;
   435	
   436	  return (
   437	    <mesh rotation={[-Math.PI / 2, 0, 0]} position={[0, -0.9, -14]}>
   438	      <circleGeometry args={[20, 96]} />
   439	      <shaderMaterial
   440	        ref={materialRef}
   441	        transparent
   442	        depthWrite={false}
   443	        uniforms={{
   444	          uTime: { value: 0 },
   445	          uVisible: { value: phase === "home" ? 1 : 0 },
   446	        }}
   447	        vertexShader={`
   448	          varying vec2 vUv;
   449	          void main() {
   450	            vUv = uv;
   451	            gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   452	          }
   453	        `}
   454	        fragmentShader={`
   455	          uniform float uTime;
   456	          uniform float uVisible;
   457	          varying vec2 vUv;
   458	
   459	          void main() {
   460	            vec2 uv = vUv - 0.5;
   461	            float d = length(uv);
   462	            float ring = smoothstep(0.62, 0.18, d);
   463	            float outer = smoothstep(0.92, 0.48, d);
   464	            float drift = 0.5 + 0.5 * sin(uTime * 0.12 + uv.x * 8.0);
   465	            vec3 inner = vec3(0.03, 0.08, 0.16);
   466	            vec3 outerColor = vec3(0.01, 0.03, 0.08);
   467	            vec3 color = mix(outerColor, inner, ring * (0.72 + drift * 0.12));
   468	            float alpha = outer * 0.36 * uVisible;
   469	            gl_FragColor = vec4(color, alpha);
   470	          }
   471	        `}
   472	      />
   473	    </mesh>
   474	  );
   475	}

   477	function HomeDust({
   478	  phase,
   479	  transitionMode,
   480	  transitionProgress,
   481	}: {
   482	  phase: ScenePhase;
   483	  transitionMode: TransitionMode;
   484	  transitionProgress: number;
   485	}) {
   486	  const dust = useMemo(() => {
   487	    const rand = seededRandom(314159);
   488	    const out: Array<[number, number, number, number]> = [];
   489	    for (let i = 0; i < 28; i += 1) {
   490	      out.push([
   491	        (rand() - 0.5) * 18,
   492	        0.2 + rand() * 5.2,
   493	        -10 - rand() * 18,
   494	        0.015 + (i % 5) * 0.005,
   495	      ]);
   496	    }
   497	    return out;
   498	  }, []);
   499	
   500	  let opacityScale = phase === "home" ? 1 : 0;
   501	  if (transitionMode === "homeToLifemap") {
   502	    opacityScale = 1 - easeInOutCubic(clamp01(transitionProgress / 0.34));
   503	  } else if (transitionMode === "lifemapToHome") {
   504	    opacityScale = easeInOutCubic(clamp01((transitionProgress - 0.50) / 0.50));
   505	  }
   506	
   507	  if (opacityScale <= 0.001) return null;
   508	
   509	  return (
   510	    <>
   511	      {dust.map((d, i) => (
   512	        <mesh key={`dust-${i}`} position={[d[0], d[1], d[2]]}>
   513	          <sphereGeometry args={[d[3], 10, 10]} />
   514	          <meshBasicMaterial
   515	            color="#9fb8d8"
   516	            transparent
   517	            opacity={(0.08 + (i % 4) * 0.02) * opacityScale}
   518	          />
   519	        </mesh>
   520	      ))}
   521	    </>
   522	  );
   523	}

   535	}) {
   536	  const groupRef = useRef<THREE.Group | null>(null);
   537	
   538	  useFrame((state) => {
   539	    if (!groupRef.current || !selected) return;
   540	
   541	    const t = state.clock.elapsedTime;
   542	    const active =
   543	      phase === "focus" ||
   544	      phase === "replay" ||
   545	      transitionMode === "lifemapToFocus" ||
   546	      transitionMode === "focusToLifemap" ||
   547	      transitionMode === "focusToReplay" ||
   548	      transitionMode === "replayToFocus";
   549	
   550	    let alpha = active ? 1 : 0;
   551	    if (transitionMode === "lifemapToFocus") alpha = easeInOutCubic(transitionProgress);
   552	    if (transitionMode === "focusToLifemap") alpha = 1 - easeInOutCubic(transitionProgress);
   553	
   554	    const replayPulse =
   555	      transitionMode === "focusToReplay"
   556	        ? 1 + easeInOutCubic(clamp01((transitionProgress - 0.08) / 0.34)) * 0.18
   557	        : phase === "replay"
   558	          ? 1.12
   559	          : 1;
   560	
   561	    const pulse = 1 + Math.sin(t * 1.25) * 0.02;
   562	    const [sx, sy, sz] = selected.position;
   563	
   564	    groupRef.current.position.set(sx, sy, sz);
   565	    groupRef.current.scale.setScalar(Math.max(0.001, alpha * pulse * replayPulse));
   566	  });
   567	
   568	  if (!selected) return null;
   569	
   570	  const visible =
   571	    phase === "focus" ||
   572	    phase === "replay" ||
   573	    transitionMode === "lifemapToFocus" ||
   574	    transitionMode === "focusToLifemap" ||
   575	    transitionMode === "focusToReplay" ||
   576	    transitionMode === "replayToFocus";
   577	
   578	  if (!visible) return null;
   579	
   580	  const isReplayLike =
   581	    phase === "replay" ||
   582	    transitionMode === "focusToReplay" ||
   583	    transitionMode === "replayToFocus";
   584	
   585	  return (
   586	    <group ref={groupRef}>
   587	      <mesh>
   588	        <sphereGeometry args={[0.48, 64, 64]} />
   589	        <meshBasicMaterial color="#eef5ff" transparent opacity={0.96} />
   590	      </mesh>
   591	
   592	      <mesh scale={1.22}>
   593	        <sphereGeometry args={[0.48, 56, 56]} />
   594	        <meshBasicMaterial color="#a7c7ff" transparent opacity={isReplayLike ? 0.22 : 0.16} depthWrite={false} />
   595	      </mesh>
   596	
   597	      <mesh scale={1.78}>
   598	        <sphereGeometry args={[0.48, 48, 48]} />
   599	        <meshBasicMaterial color="#5d8eff" transparent opacity={isReplayLike ? 0.12 : 0.075} depthWrite={false} />
   600	      </mesh>
   601	
   602	      <mesh scale={[1.0, 1.0, 1.25]}>
   603	        <sphereGeometry args={[0.18, 48, 48]} />
   604	        <meshBasicMaterial color="#ffffff" transparent opacity={0.70} depthWrite={false} />
   605	      </mesh>
   606	    </group>

   610	function Starfield({
   611	  phase,
   612	  transitionMode,
   613	  transitionProgress,
   614	  stars,
   615	  selected,
   616	  hoveredId,
   617	  onHover,
   618	  onUnhover,
   619	  onSelect,
   620	}: {
   621	  phase: ScenePhase;
   622	  transitionMode: TransitionMode;
   623	  transitionProgress: number;
   624	  stars: StarPoint[];
   625	  selected: StarPoint | null;
   626	  hoveredId: string | null;
   627	  onHover: (id: string) => void;
   628	  onUnhover: () => void;
   629	  onSelect: (star: StarPoint) => void;
   630	}) {
   631	  let fieldOpacity = phase === "home" ? 0 : 1;
   632	
   633	  if (transitionMode === "homeToLifemap") {
   634	    fieldOpacity = easeInOutCubic(clamp01((transitionProgress - 0.50) / 0.50));
   635	  } else if (transitionMode === "lifemapToHome") {
   636	    fieldOpacity = 1 - easeInOutCubic(clamp01(transitionProgress / 0.28));
   637	  }
   638	
   639	  if (fieldOpacity <= 0.001) return null;
   640	
   641	  let focusBlend =
   642	    transitionMode === "lifemapToFocus"
   643	      ? easeInOutCubic(transitionProgress)
   644	      : transitionMode === "focusToLifemap"
   645	        ? 1 - easeInOutCubic(transitionProgress)
   646	        : phase === "focus" || phase === "replay" || transitionMode === "focusToReplay" || transitionMode === "replayToFocus"
   647	          ? 1
   648	          : 0;
   649	
   650	  const replayThreshold =
   651	    transitionMode === "focusToReplay"
   652	      ? easeInOutCubic(clamp01((transitionProgress - 0.12) / 0.58))
   653	      : transitionMode === "replayToFocus"
   654	        ? 1 - easeInOutCubic(clamp01((transitionProgress - 0.12) / 0.58))
   655	        : phase === "replay"
   656	          ? 1
   657	          : 0;
   658	
   659	  focusBlend = Math.max(focusBlend, replayThreshold);
   660	
   661	  return (
   662	    <>
   663	      {stars.map((star) => {
   664	        const isSelected = selected?.id === star.id;
   665	        const isHovered = hoveredId === star.id;
   666	
   667	        const dim = focusBlend > 0 ? (isSelected ? 0 : mix(0.82, 0.04, focusBlend)) : 0.82;
   668	        const hoverScale = isHovered && focusBlend < 0.1 ? 1.15 : 1;
   669	        const scale = hoverScale;
   670	
   671	        const [sx, sy, sz] = star.position;
   672	        let px = sx;
   673	        let py = sy;
   674	        let pz = sz;
   675	
   676	        if (selected && !isSelected && focusBlend > 0.001) {
   677	          const dx = sx - selected.position[0];
   678	          const dy = sy - selected.position[1];
   679	          const drift = 1 + focusBlend * 0.34;
   680	          px = selected.position[0] + dx * drift;
   681	          py = selected.position[1] + dy * drift * 1.12;
   682	          pz = sz - focusBlend * 8.0;
   683	        }
   684	
   685	        if (selected && isSelected && focusBlend > 0.001) {
   686	          return null;
   687	        }
   688	
   689	        const glowOpacity = (isHovered ? 0.08 : 0.04) * dim;
   690	
   691	        return (
   692	          <group key={star.id} position={[px, py, pz]}>
   693	            <mesh
   694	              onPointerOver={() => onHover(star.id)}
   695	              onPointerOut={onUnhover}
   696	              onClick={() => onSelect(star)}
   697	              scale={scale}
   698	            >
   699	              <sphereGeometry args={[star.size, 20, 20]} />
   700	              <meshBasicMaterial
   701	                color="#dbe8ff"
   702	                transparent
   703	                opacity={Math.min(1, dim * (0.45 + star.importance * 0.55) * fieldOpacity)}
   704	              />
   705	            </mesh>
   706	            <mesh scale={1.75 * scale}>
   707	              <sphereGeometry args={[star.size, 16, 16]} />
   708	              <meshBasicMaterial
   709	                color="#7ca7df"
   710	                transparent
   711	                opacity={glowOpacity * fieldOpacity}
   712	              />
   713	            </mesh>
   714	          </group>
   715	        );
   716	      })}
   717	    </>
   718	  );
   719	}

   721	function ReplayField({
   722	  active,
   723	  transitionMode,
   724	  transitionProgress,
   725	  selected,
   726	}: {
   727	  active: boolean;
   728	  transitionMode: TransitionMode;
   729	  transitionProgress: number;
   730	  selected: StarPoint | null;
   731	}) {
   732	  const groupRef = useRef<THREE.Group | null>(null);
   733	
   734	  useFrame((state) => {
   735	    if (!groupRef.current || !selected) return;
   736	    groupRef.current.rotation.z = Math.sin(state.clock.elapsedTime * 0.18) * 0.06;
   737	
   738	    let scale = active ? 1 : 0;
   739	    if (transitionMode === "focusToReplay") {
   740	      scale = easeInOutCubic(clamp01((transitionProgress - 0.12) / 0.88));
   741	    } else if (transitionMode === "replayToFocus") {
   742	      scale = 1 - easeInOutCubic(clamp01((transitionProgress - 0.10) / 0.90));
   743	    }
   744	
   745	    groupRef.current.scale.setScalar(Math.max(0.001, scale));
   746	  });
   747	
   748	  if (!selected) return null;
   749	
   750	  const showing =
   751	    active ||
   752	    transitionMode === "focusToReplay" ||
   753	    transitionMode === "replayToFocus";
   754	
   755	  if (!showing) return null;
   756	
   757	  const [sx, sy, sz] = selected.position;
   758	
   759	  return (
   760	    <group ref={groupRef} position={[sx, sy, sz - 0.2]}>
   761	      {Array.from({ length: 52 }).map((_, i) => {
   762	        const angle = (i / 52) * Math.PI * 2;
   763	        const radius = 0.54 + (i % 8) * 0.19;
   764	        const x = Math.cos(angle) * radius;
   765	        const y = Math.sin(angle) * radius * 0.34;
   766	        const z = -i * 0.42;
   767	        const stretch = 1 + (i % 5) * 0.08;
   768	
   769	        return (
   770	          <mesh key={`replay-${i}`} position={[x, y, z]} scale={[1, 1, stretch]}>
   771	            <sphereGeometry args={[0.028 + (i % 5) * 0.008, 10, 10]} />
   772	            <meshBasicMaterial color="#cfe2ff" transparent opacity={0.08 + (i % 6) * 0.024} />
   773	          </mesh>
   774	        );
   775	      })}
   776	    </group>
   777	  );
   778	}

   937	function SceneContent(): JSX.Element {
   938	  const stars = useMemo(() => buildStars(), []);
   939	  const [phase, setPhase] = useState<ScenePhase>("home");
   940	  const [selected, setSelected] = useState<StarPoint | null>(null);
   941	  const [hoveredId, setHoveredId] = useState<string | null>(null);
   942	  const [transitionMode, setTransitionMode] = useState<TransitionMode>("idle");
   943	  const [transitionProgress, setTransitionProgress] = useState(0);
   944	
   945	  const queueRef = useRef<{ phase: ScenePhase; clearSelected?: boolean } | null>(null);
   946	
   947	  const beginTransition = useCallback(
   948	    (
   949	      mode: TransitionMode,
   950	      nextPhase: ScenePhase,
   951	      opts?: { star?: StarPoint | null; clearSelected?: boolean }
   952	    ) => {
   953	      if (transitionMode !== "idle") return;
   954	
   955	      if (opts && Object.prototype.hasOwnProperty.call(opts, "star")) {
   956	        setSelected(opts?.star ?? null);
   957	      }
   958	
   959	      queueRef.current = {
   960	        phase: nextPhase,
   961	        clearSelected: opts?.clearSelected,
   962	      };
   963	
   964	      setTransitionProgress(0);
   965	      setTransitionMode(mode);
   966	    },
   967	    [transitionMode]
   968	  );
   969	
   970	  useFrame((_, delta) => {
   971	    if (transitionMode === "idle") return;
   972	
   973	    const duration = getTransitionDuration(transitionMode);
   974	    if (duration <= 0) return;
   975	
   976	    setTransitionProgress((prev) => {
   977	      const next = clamp01(prev + delta / duration);
   978	
   979	      if (next >= 1) {
   980	        const queued = queueRef.current;
   981	        queueRef.current = null;
   982	        setTransitionMode("idle");
   983	        setTransitionProgress(0);
   984	
   985	        if (queued) {
   986	          setPhase(queued.phase);
   987	          if (queued.clearSelected) {
   988	            setSelected(null);
   989	          }
   990	        }
   991	
   992	        return 1;
   993	      }
   994	
   995	      return next;
   996	    });
   997	  });
   998	
   999	  const onSelectStar = useCallback(
  1000	    (star: StarPoint) => {
  1001	      if (phase !== "lifemap" || transitionMode !== "idle") return;
  1002	      beginTransition("lifemapToFocus", "focus", { star });
  1003	    },
  1004	    [phase, transitionMode, beginTransition]

  1013	          beginTransition("replayToFocus", "focus");

  1015	        }

  1017	        if (phase === "focus") {
  1018	          beginTransition("focusToLifemap", "lifemap");
  1019	          return;
  1020	        }

  1023	          beginTransition("lifemapToHome", "home", { clearSelected: true });
  1024	        }
  1025	
  1026	        return;
  1027	      }
  1028	
  1029	      if (event.key === "Enter" && phase === "focus" && selected) {
  1030	        beginTransition("focusToReplay", "replay");
  1031	      }
  1032	    };
  1033	
  1034	    window.addEventListener("keydown", onKey);
  1035	    return () => window.removeEventListener("keydown", onKey);
  1036	  }, [phase, selected, transitionMode, beginTransition]);

  1049	    orbIntensity = 1.05 * (1 - fade);

  1057	    orbOpacity = rise;
  1058	    orbY = mix(-3.4, 0.35, rise);
  1059	    orbZ = mix(8.8, 0.7, rise);

  1068	        : transitionMode === "homeToLifemap"
  1069	          ? 1.1 * (1 - easeInOutCubic(clamp01(transitionProgress / 0.30)))
  1070	          : 0.10;

  1079	      <pointLight
  1080	        position={[0, 1.6, 1.8]}
  1081	        intensity={pointLightIntensity}
  1082	        distance={32}
  1083	        color="#bdd6ff"
  1084	      />
  1085	
  1086	      <SkyDome
  1087	        phase={phase}
  1088	        transitionMode={transitionMode}
  1089	        transitionProgress={transitionProgress}
  1090	      />
  1091	
  1092	      <CameraRig
  1093	        phase={phase}
  1094	        transitionMode={transitionMode}
  1095	        transitionProgress={transitionProgress}
  1096	        selected={selected}
  1097	      />
  1098	
  1099	      <Ground
  1100	        phase={phase}
  1101	        transitionMode={transitionMode}
  1102	        transitionProgress={transitionProgress}
  1103	      />
  1104	
  1105	      <HomeDust
  1106	        phase={phase}
  1107	        transitionMode={transitionMode}
  1108	        transitionProgress={transitionProgress}
  1109	      />
  1110	
  1111	      <group
  1112	        visible={
  1113	          phase === "home" ||
  1114	          transitionMode === "homeToLifemap" ||
  1115	          transitionMode === "lifemapToHome"
  1116	        }
  1117	      >

     1	"use client";
     2	
     3	import { useFrame } from "@react-three/fiber";
     4	import { useRef } from "react";
     5	import { Vector3 } from "three";
     6	import type { StarPoint } from "./sceneState";
     7	
     8	type CameraPhase =
     9	  | "home"
    10	  | "lifemap"
    11	  | "focus"
    12	  | "replay"
    13	  | "focus_travel"
    14	  | "focus_arrive"
    15	  | "returningHome"
    16	  | "ascent"
    17	  | string;
    18	
    19	type Props = {
    20	  phase: CameraPhase;
    21	  progress?: number;
    22	  selected?: StarPoint | null;
    23	  selectedStar?: StarPoint | null;
    24	};
    25	
    26	const HOME_POS = new Vector3(0, 0.92, 7.6);
    27	const HOME_LOOK = new Vector3(0, 0.88, 0);
    28	const LIFEMAP_POS = new Vector3(0, 2.95, 16.2);
    29	const LIFEMAP_LOOK = new Vector3(0, 2.45, -21.5);
    30	
    31	export default function CameraRig({
    32	  phase,
    33	  progress = 1,
    34	  selected = null,
    35	  selectedStar = null,
    36	}: Props) {
    37	  const activeStar = selectedStar ?? selected ?? null;
    38	  const target = useRef(HOME_POS.clone());
    39	  const lookAt = useRef(HOME_LOOK.clone());
    40	  const fallbackDir = useRef(new Vector3(0, 0.18, 1));
    41	
    42	  useFrame((state) => {
    43	    const cam = state.camera;
    44	    const tNow = state.clock.getElapsedTime();
    45	
    46	    let desiredPos = HOME_POS.clone();
    47	    let desiredLook = HOME_LOOK.clone();
    48	    let lerpFactor = 0.046;
    49	
    50	    if (phase === "home") {
    51	      desiredPos.x += Math.sin(tNow * 0.18) * 0.05;
    52	      desiredPos.y += Math.sin(tNow * 0.22) * 0.035;
    53	      desiredLook.x += Math.sin(tNow * 0.14) * 0.025;
    54	      desiredLook.y += Math.sin(tNow * 0.18) * 0.02;
    55	    }
    56	
    57	    if (phase === "lifemap") {
    58	      desiredPos = LIFEMAP_POS.clone();
    59	      desiredLook = LIFEMAP_LOOK.clone();
    60	      desiredPos.x += Math.sin(tNow * 0.12) * 0.14;
    61	      desiredPos.y += Math.sin(tNow * 0.16) * 0.08;
    62	      desiredLook.x += Math.sin(tNow * 0.10) * 0.08;
    63	      desiredLook.y += Math.sin(tNow * 0.13) * 0.04;
    64	      lerpFactor = 0.055;
    65	    }
    66	
    67	    if (activeStar) {
    68	      const starPos = new Vector3(
    69	        activeStar.position[0],
    70	        activeStar.position[1],
    71	        activeStar.position[2]
    72	      );
    73	
    74	      let offset = fallbackDir.current.clone();
    75	      const toCamera = new Vector3().subVectors(cam.position, starPos);
    76	      if (toCamera.lengthSq() > 0.0001) {
    77	        offset = toCamera.normalize();
    78	        fallbackDir.current.copy(offset);
    79	      }
    80	
    81	      const t = Math.max(0, Math.min(1, progress));
    82	
    83	      if (phase === "focus" || phase === "focus_travel" || phase === "focus_arrive") {
    84	        const dist = 4.35 - t * 1.0;
    85	        desiredPos = starPos.clone().add(offset.clone().multiplyScalar(dist));
    86	        desiredPos.x += Math.sin(tNow * 0.2) * 0.03;
    87	        desiredPos.y += 0.3 + Math.sin(tNow * 0.24) * 0.025;
    88	        desiredLook = starPos.clone();
    89	        desiredLook.y += 0.1;
    90	        lerpFactor = 0.07;
    91	      } else if (phase === "replay") {
    92	        const dist = 2.18 - t * 0.42;
    93	        desiredPos = starPos.clone().add(offset.clone().multiplyScalar(dist));
    94	        desiredPos.x += Math.sin(tNow * 0.24) * 0.02;
    95	        desiredPos.y += 0.16 + Math.sin(tNow * 0.28) * 0.018;
    96	        desiredLook = starPos.clone();
    97	        desiredLook.y += 0.04;
    98	        lerpFactor = 0.082;
    99	      }
   100	    }
   101	
   102	    target.current.lerp(desiredPos, lerpFactor);
   103	    lookAt.current.lerp(desiredLook, lerpFactor * 0.9);
   104	
   105	    cam.position.copy(target.current);
   106	    cam.lookAt(lookAt.current);
   107	  });
   108	
   109	  return null;
   110	}
