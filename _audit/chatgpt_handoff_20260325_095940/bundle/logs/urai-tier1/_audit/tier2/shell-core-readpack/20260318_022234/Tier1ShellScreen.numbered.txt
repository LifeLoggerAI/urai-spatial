    66	function seededDust(count: number) {
    67	  let seed = 1337;
    68	  const next = () => {
    69	    seed = (seed * 1664525 + 1013904223) >>> 0;
    70	    return seed / 4294967296;
    71	  };
    72	  return Array.from({ length: count }, (_, index) => ({
    73	    id: `dust-${index}`,
    74	    x: Math.round(next() * 1000) / 10,
    75	    y: Math.round(next() * 1000) / 10,
    76	    o: 0.15 + next() * 0.6,
    77	    s: 1 + next() * 2.4,
    78	  }));
    79	}
