     1	"use client";
     2	import { useXrSessionStore } from "@/spatial/xr/xrSessionStore";
     3	
     4	import { useEffect, useMemo, useRef } from "react";
     5	import { useFrame, useThree } from "@react-three/fiber";
     6	import { Vector3 } from "three";
     7	import { useSceneStore } from "../state/sceneStore";
     8	
     9	const HOME_POS = new Vector3(0, 1.15, 14.5);
    10	const HOME_LOOK = new Vector3(0, 0.35, 0);
    11	
    12	const MAP_POS = new Vector3(0, 2.8, 20.5);
    13	const MAP_LOOK = new Vector3(0, 0, 0);
    14	
    15	function smooth(delta: number, speed: number) {
    16	  return 1 - Math.exp(-delta * speed);
    17	}
    18	
    19	export default function CameraRig() {
    20	  const camera = useThree((state) => state.camera);
    21	
    22	  const mode = useSceneStore((state) => state.mode);
    23	  const selectedStar = useSceneStore((state) => state.selectedStar);
    24	  const replayEnteredAt = 0;
    25	
    26	  const posRef = useRef(new Vector3());
    27	  const lookRef = useRef(new Vector3());
    28	
    29	  const selectedTarget = useMemo(() => {
    30	    if (!selectedStar) return new Vector3(0, 0, 0);
    31	    return new Vector3(
    32	      selectedStar.position[0],
    33	      selectedStar.position[1],
    34	      selectedStar.position[2]
    35	    );
    36	  }, [selectedStar]);
    37	
    38	  useEffect(() => {
    39	    camera.position.copy(HOME_POS);
    40	    camera.lookAt(HOME_LOOK);
    41	    posRef.current.copy(HOME_POS);
    42	    lookRef.current.copy(HOME_LOOK);
    43	  }, [camera]);
    44	
    45	  useFrame((state, delta) => {
    46	    let desiredPos = HOME_POS.clone();
    47	    let desiredLook = HOME_LOOK.clone();
    48	
    49	    if (mode === "lifemap") {
    50	      desiredPos = MAP_POS.clone();
    51	      desiredLook = MAP_LOOK.clone();
    52	    }
    53	
    54	    if ((mode === "focus" || mode === "replay") && selectedStar) {
    55	      const focusLift = 1.30 + selectedStar.size * 0.12;
    56	      const focusDistance = 6.4 + Math.min(selectedStar.size * 0.34, 1.35);
    57	
    58	      desiredLook = selectedTarget.clone();
    59	      desiredPos = selectedTarget.clone().add(new Vector3(1.1, focusLift, focusDistance));
    60	    }
    61	
    62	    if (mode === "replay" && selectedStar) {
    63	      const enteredAt = replayEnteredAt ?? Date.now();
    64	      const replayT = Math.max(0, (Date.now() - enteredAt) / 1000);
    65	
    66	      const settle = Math.min(replayT / 1.4, 1);
    67	      const orbitAngle = replayT * 0.58;
    68	      const baseRadius = 6.0 + Math.min(selectedStar.size * 0.40, 1.4);
    69	      const radius = baseRadius - settle * 0.4;
    70	
    71	      const x = Math.cos(orbitAngle) * radius;
    72	      const z = Math.sin(orbitAngle) * radius;
    73	      const y =
    74	        0.95 +
    75	        Math.sin(replayT * 0.7) * 0.22 +
    76	        Math.cos(replayT * 0.23) * 0.08;
    77	
    78	      desiredLook = selectedTarget.clone().add(
    79	        new Vector3(
    80	          Math.sin(replayT * 0.21) * 0.1,
    81	          0.02 + Math.cos(replayT * 0.33) * 0.04,
    82	          0
    83	        )
    84	      );
    85	
    86	      desiredPos = selectedTarget.clone().add(new Vector3(x, y, z));
    87	    }
    88	
    89	    const positionSpeed =
    90	      mode === "replay" ? 2.2 : mode === "focus" ? 3.2 : 2.6;
    91	    const lookSpeed =
    92	      mode === "replay" ? 3.1 : mode === "focus" ? 4.2 : 3.0;
    93	
    94	    posRef.current.lerp(desiredPos, smooth(delta, positionSpeed));
    95	    lookRef.current.lerp(desiredLook, smooth(delta, lookSpeed));
    96	
    97	    camera.position.copy(posRef.current);
    98	    camera.lookAt(lookRef.current);
    99	  });
   100	
   101	  return null;
   102	}
